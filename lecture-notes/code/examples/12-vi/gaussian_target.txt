>>> import numpy as np
>>> from vi import gaussian_vi, gaussian_kl, factor
>>> C = np.array([[1.0, 0.9], [0.9, 1.0]])                            #<1>
>>> Ci = np.linalg.inv(C)
>>> log_p_grad = lambda th: (-0.5 * th @ Ci @ th, -Ci @ th)
>>> m, rho, off, hist = gaussian_vi(log_p_grad, [2.0, -1.0], [0.0, 0.0],
...                                 4000, np.random.default_rng(0))
>>> R = factor(rho, off, False)
>>> print(np.round(m, 1) + 0.0, np.round(R @ R.T, 1).tolist())         #<2>
[0. 0.] [[1.0, 0.9], [0.9, 1.0]]
