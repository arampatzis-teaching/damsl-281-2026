>>> import numpy as np
>>> u = np.array([0.0, 1.0, 4.0, 9.0, 16.0])
>>> dx = 1.0
>>> lap = np.empty_like(u)
>>> for j in range(u.size):                                        #<1>
...     lap[j] = (u[(j + 1) % u.size] - 2 * u[j] + u[j - 1]) / dx**2
>>> lap_roll = (np.roll(u, -1) - 2 * u + np.roll(u, 1)) / dx**2   #<2>
>>> print(np.allclose(lap, lap_roll))
True
>>> print(np.shares_memory(np.roll(u, -1), u))                    #<3>
False
>>> print(np.shares_memory(u[2:], u))                             #<4>
True
>>> out = np.empty_like(u)                                        #<5>
>>> out[1:-1] = (u[2:] - 2 * u[1:-1] + u[:-2]) / dx**2
>>> out[0] = (u[1] - 2 * u[0] + u[-1]) / dx**2                    #<6>
>>> out[-1] = (u[0] - 2 * u[-1] + u[-2]) / dx**2
>>> print(np.allclose(out, lap_roll))
True
