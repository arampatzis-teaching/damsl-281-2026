>>> import numpy as np
>>> from pod import pod, floor
>>> rng = np.random.default_rng(0)
>>> Y = rng.standard_normal((50, 8)) @ rng.standard_normal((8, 30))    #<1>
>>> V, s = pod(Y, 3)
>>> print(np.allclose(V.T @ V, np.eye(3)))
True
>>> error = np.linalg.norm(Y - V @ (V.T @ Y)) / np.linalg.norm(Y)      #<2>
>>> print(np.isclose(error, floor(s)[3]))
True
>>> print(floor(s)[8] < 1e-12)                                         #<3>
True
>>> W = np.diag(rng.uniform(0.5, 2.0, 50))                             #<4>
>>> V, s = pod(Y, 3, W)
>>> print(np.allclose(V.T @ W @ V, np.eye(3)))
True
