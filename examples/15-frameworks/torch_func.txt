>>> import torch
>>> from torch.func import grad, vmap, jacrev
>>> def f(x):
...     return torch.sin(x**2)
>>> df = grad(f)                                        #<1>
>>> print(f"{df(torch.tensor(1.0)):.4f}")
1.0806
>>> d2f = grad(grad(f))                                 #<2>
>>> print(f"{d2f(torch.tensor(1.0)):.4f}")
-2.2853
>>> xs = torch.tensor([1.0, 2.0, 3.0])
>>> print(vmap(df)(xs))                                 #<3>
tensor([ 1.0806, -2.6146, -5.4668])
>>> def g(x):
...     return torch.stack([x[0] * x[1], torch.sin(x[0] + x[1])])
>>> print(jacrev(g)(torch.tensor([0.5, 2.0])))          #<4>
tensor([[ 2.0000,  0.5000],
        [-0.8011, -0.8011]])
