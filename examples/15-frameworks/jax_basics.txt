>>> import jax
>>> import jax.numpy as jnp
>>> def f(x):
...     return jnp.sin(x**2)
>>> print(f"{jax.grad(f)(1.0):.4f}")                    #<1>
1.0806
>>> slopes = jax.vmap(jax.grad(f))(jnp.array([1.0, 2.0, 3.0]))   #<2>
>>> print([f"{v:.4f}" for v in slopes])
['1.0806', '-2.6146', '-5.4668']
>>> fast = jax.jit(jax.grad(f))                         #<3>
>>> print(f"{fast(1.0):.4f}")
1.0806
>>> x = jnp.arange(4.0)

[0. 1. 2. 3.] [5. 1. 2. 3.]
