>>> import torch
>>> from pinn import MLP, d, train
>>> _ = torch.manual_seed(0)
>>> u0, T = 0.1, 5.0
>>> net = MLP([1, 32, 32, 1])
>>> u = lambda t: u0 + (1 - torch.exp(-t)) * net(t)
>>> def loss_upto(t_stage):                                     #<1>
...     def loss():
...         t = (t_stage * torch.rand(512, 1)).requires_grad_(True)
...         ut = u(t)
...         return ((d(ut, t) - (ut - ut**3))**2).mean()
...     return loss
>>> history = []
>>> for stage in range(1, 6):                                   #<2>
...     history += train(loss_upto(stage * T / 5), net.parameters(), 1000)
>>> print(f"{history[-1]:.1e}")
1.0e-05
