>>> import numpy as np, torch
>>> from pinn import Constrained, points
>>> _ = torch.manual_seed(0)
>>> u0 = lambda x: x**2 * torch.cos(np.pi * x)                  #<1>
>>> net = Constrained(u0, [64, 64])                             #<2>
>>> X = points([(-1.0, 1.0), (0.0, 5.0)], 4)
>>> print(sum(p.numel() for p in net.parameters()))
4481
>>> start = torch.cat([X[:, :1], torch.zeros_like(X[:, :1])], 1)
>>> print(torch.allclose(net(start), u0(X[:, :1])))             #<3>
True
>>> t = torch.rand(4, 1)
>>> left = torch.cat([-torch.ones_like(t), t], 1)
>>> right = torch.cat([torch.ones_like(t), t], 1)
>>> print(torch.allclose(net(left), net(right), atol=1e-6))     #<4>
True
